from __future__ import annotations

import math
from typing import Dict, Optional

# ============================================================
# Default 3PL IRT Parameters
# ============================================================
# a = discrimination
# b = difficulty
# c = pseudo-guessing (near 0 for Likert scale)
#
# Higher weight -> harder + more discriminating
# q8 hardest and strongest discriminator
# q1 easiest and weakest discriminator
# ============================================================

IRT_PARAMS = {
    "q1": {"a": 0.65, "b": -2.0, "c": 0.01},  # Innovation Drive
    "q2": {"a": 0.80, "b": -1.4, "c": 0.01},
    "q3": {"a": 0.95, "b": -0.8, "c": 0.01},
    "q4": {"a": 1.10, "b": -0.2, "c": 0.01},
    "q5": {"a": 1.25, "b": 0.5, "c": 0.01},
    "q6": {"a": 1.45, "b": 1.0, "c": 0.01},
    "q7": {"a": 1.70, "b": 1.6, "c": 0.01},
    "q8": {"a": 2.00, "b": 2.2, "c": 0.01},  # Chronotype Sync
}

THETA_GRID = [x / 10 for x in range(-40, 41)]  # -4.0 to +4.0
PRIOR_SD = 1.0
STOP_SE = 0.35


# ============================================================
# Utility Functions
# ============================================================

def sigmoid(x: float) -> float:
    return 1.0 / (1.0 + math.exp(-x))


def irt_3pl_probability(theta: float, a: float, b: float, c: float) -> float:
    return c + (1.0 - c) * sigmoid(a * (theta - b))


def normalize_likert_response(answer: int) -> float:
    """
    Maps Likert 1-5 to probability-like signal in [0,1].
    """
    if answer < 1 or answer > 5:
        raise ValueError("Answers must be integers from 1 to 5.")
    return (answer - 1) / 4.0


def response_likelihood(
    theta: float,
    answer: int,
    a: float,
    b: float,
    c: float,
) -> float:
    """
    Treat normalized Likert response as soft correctness signal.
    """
    p = irt_3pl_probability(theta, a, b, c)
    r = normalize_likert_response(answer)

    p = min(max(p, 1e-9), 1 - 1e-9)

    return (p ** r) * ((1 - p) ** (1 - r))


def normal_prior(theta: float, mean: float = 0.0, sd: float = PRIOR_SD) -> float:
    coef = 1.0 / (sd * math.sqrt(2 * math.pi))
    exp_term = math.exp(-0.5 * ((theta - mean) / sd) ** 2)
    return coef * exp_term


# ============================================================
# EAP Theta Estimation
# ============================================================

def eap_theta_estimation(current_answers: Dict[str, int]) -> tuple[float, float]:
    """
    Returns:
        theta_hat, se_theta
    """

    posterior = []

    for theta in THETA_GRID:
        likelihood = 1.0

        for qid, answer in current_answers.items():
            params = IRT_PARAMS[qid]

            likelihood *= response_likelihood(
                theta=theta,
                answer=answer,
                a=params["a"],
                b=params["b"],
                c=params["c"],
            )

        posterior_value = likelihood * normal_prior(theta)
        posterior.append(posterior_value)

    total = sum(posterior)

    if total == 0:
        return 0.0, float("inf")

    posterior = [p / total for p in posterior]

    theta_hat = sum(t * p for t, p in zip(THETA_GRID, posterior))

    variance = sum(
        ((t - theta_hat) ** 2) * p
        for t, p in zip(THETA_GRID, posterior)
    )

    se_theta = math.sqrt(max(variance, 1e-9))

    return theta_hat, se_theta


# ============================================================
# Fisher Information
# ============================================================

def fisher_information(theta: float, a: float, b: float, c: float) -> float:
    p = irt_3pl_probability(theta, a, b, c)

    p = min(max(p, 1e-9), 1 - 1e-9)

    numerator = (a ** 2) * ((p - c) ** 2)
    denominator = ((1 - c) ** 2) * p * (1 - p)

    return numerator / denominator


# ============================================================
# Public API
# ============================================================

def cat_estimated_theta(current_answers: Dict[str, int]) -> dict:
    theta, se = eap_theta_estimation(current_answers)

    answered = len(current_answers)
    remaining = len(IRT_PARAMS) - answered

    rationale = (
        f"EAP theta estimated using {answered} answered items. "
        f"Current theta={theta:.3f}, SE={se:.3f}, "
        f"{remaining} items remaining."
    )

    return {
        "theta": round(theta, 4),
        "se": round(se, 4),
        "rationale": rationale,
    }


def cat_select_next_question(
    current_answers: Dict[str, int]
) -> Optional[str]:
    """
    Select next question using maximum Fisher Information rule.
    Stops when:
    - SE(theta) < STOP_SE
    - all questions answered
    """

    unanswered = [
        qid for qid in IRT_PARAMS
        if qid not in current_answers
    ]

    if not unanswered:
        return None

    theta, se = eap_theta_estimation(current_answers)

    if se < STOP_SE:
        return None

    best_question = None
    best_information = -1.0

    for qid in unanswered:
        params = IRT_PARAMS[qid]

        info = fisher_information(
            theta=theta,
            a=params["a"],
            b=params["b"],
            c=params["c"],
        )

        if info > best_information:
            best_information = info
            best_question = qid

    return best_question


# ============================================================
# Optional Full Response Helper
# ============================================================

def cat_step(current_answers: Dict[str, int]) -> dict:
    theta_data = cat_estimated_theta(current_answers)

    next_question = cat_select_next_question(current_answers)

    return {
        "next_question_id": next_question,
        "estimated_theta": theta_data["theta"],
        "se_theta": theta_data["se"],
        "rationale": theta_data["rationale"],
    }


# ============================================================
# Example
# ============================================================

if __name__ == "__main__":
    answers = {
        "q1": 5,
        "q2": 4,
        "q3": 3,
    }

    result = cat_step(answers)

    print(result)

from __future__ import annotations

import base64
from typing import Optional

from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC


_SALT = b"gitsyntropy-token-encryption"
_ITERATIONS = 390_000
_KEY_LENGTH = 32


def _derive_key(secret: str) -> bytes:
    if not isinstance(secret, str):
        raise TypeError("secret must be a string")

    secret_bytes = secret.encode("utf-8")

    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=_KEY_LENGTH,
        salt=_SALT,
        iterations=_ITERATIONS,
    )

    derived_key = kdf.derive(secret_bytes)

    return base64.urlsafe_b64encode(derived_key)


def encrypt_token(token: Optional[str], secret: str) -> Optional[str]:
    """
    Encrypt a GitHub OAuth token using Fernet symmetric encryption.

    Args:
        token: Plaintext token string
        secret: GS_JWT_SECRET value

    Returns:
        URL-safe encrypted string suitable for PostgreSQL TEXT storage
    """

    if token is None:
        return None

    if not isinstance(token, str):
        raise TypeError("token must be a string or None")

    key = _derive_key(secret)
    fernet = Fernet(key)

    encrypted = fernet.encrypt(token.encode("utf-8"))

    return encrypted.decode("utf-8")


def decrypt_token(encrypted: Optional[str], secret: str) -> Optional[str]:
    """
    Decrypt a Fernet-encrypted GitHub OAuth token.

    Args:
        encrypted: Encrypted token string from PostgreSQL
        secret: GS_JWT_SECRET value

    Returns:
        Plaintext token
    """

    if encrypted is None:
        return None

    if not isinstance(encrypted, str):
        raise TypeError("encrypted must be a string or None")

    key = _derive_key(secret)
    fernet = Fernet(key)

    decrypted = fernet.decrypt(encrypted.encode("utf-8"))

    return decrypted.decode("utf-8")

from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Dict, List, Tuple

import numpy as np
from sklearn.linear_model import LogisticRegression


@dataclass
class CalibrationConfig:
    n_dimensions: int = 8
    synthetic_samples: int = 10000
    random_seed: int = 42


class CalibrationModel:
    """
    Platt-scaled calibration model for compatibility confidence estimation.

    Inputs:
        - score_vector: ndarray shape (8,)
        - signal_coverage: float in [0, 1]

    Output:
        - calibrated_confidence: float in [0, 1]
    """

    def __init__(
        self,
        model: LogisticRegression | None = None,
        n_dimensions: int = 8,
    ) -> None:
        self.n_dimensions = n_dimensions
        self.model = model or LogisticRegression()

    # ============================================================
    # Feature Engineering
    # ============================================================

    def _feature_vector(
        self,
        score_vector: np.ndarray,
        signal_coverage: float,
    ) -> np.ndarray:
        """
        Feature construction for calibration.

        Features:
            - mean score
            - std deviation
            - min score
            - max score
            - pairwise consistency
            - coverage
            - coverage^2
            - interaction(mean * coverage)
        """

        scores = np.asarray(score_vector, dtype=np.float64)

        if scores.shape[0] != self.n_dimensions:
            raise ValueError(
                f"Expected {self.n_dimensions} dimensions, "
                f"got {scores.shape[0]}"
            )

        mean_score = np.mean(scores)
        std_score = np.std(scores)
        min_score = np.min(scores)
        max_score = np.max(scores)

        consistency = 1.0 / (1.0 + std_score)

        coverage = float(np.clip(signal_coverage, 0.0, 1.0))

        features = np.array(
            [
                mean_score,
                std_score,
                min_score,
                max_score,
                consistency,
                coverage,
                coverage**2,
                mean_score * coverage,
            ],
            dtype=np.float64,
        )

        return features

    # ============================================================
    # Synthetic Data Generation
    # ============================================================

    @classmethod
    def from_synthetic_data(
        cls,
        config: CalibrationConfig | None = None,
    ) -> "CalibrationModel":
        """
        Generate synthetic calibration data and fit the model.

        Assumptions:
            - Higher signal coverage => higher reliability
            - Higher consistency between dimensions => higher reliability
            - Extreme compatibility scores with low coverage are unreliable
            - Fully observed teams are substantially more reliable
        """

        config = config or CalibrationConfig()

        rng = np.random.default_rng(config.random_seed)

        X = []
        y = []

        for _ in range(config.synthetic_samples):
            coverage = rng.uniform(0.1, 1.0)

            observed_mask = (
                rng.uniform(0, 1, config.n_dimensions) < coverage
            )

            latent_team_quality = rng.normal(0.0, 1.0)

            score_noise = rng.normal(
                loc=0.0,
                scale=max(0.15, 1.0 - coverage),
                size=config.n_dimensions,
            )

            scores = latent_team_quality + score_noise

            scores = 1 / (1 + np.exp(-scores))

            scores = np.clip(scores, 0.0, 1.0)

            scores[~observed_mask] = 0.5

            score_std = np.std(scores)
            score_mean = np.mean(scores)

            consistency = 1.0 / (1.0 + score_std)

            certainty_signal = (
                2.4 * coverage
                + 1.6 * consistency
                + 0.5 * abs(score_mean - 0.5)
                - 1.8 * (1.0 - coverage) * abs(score_mean - 0.5)
                + rng.normal(0.0, 0.35)
            )

            true_probability = 1 / (1 + np.exp(-certainty_signal))

            success = rng.binomial(1, true_probability)

            model = cls(n_dimensions=config.n_dimensions)

            features = model._feature_vector(scores, coverage)

            X.append(features)
            y.append(success)

        X = np.asarray(X, dtype=np.float64)
        y = np.asarray(y, dtype=np.int32)

        calibrated_model = cls(
            model=LogisticRegression(max_iter=1000),
            n_dimensions=config.n_dimensions,
        )

        calibrated_model.fit(X, y)

        return calibrated_model

    # ============================================================
    # Fit
    # ============================================================

    def fit(
        self,
        X: np.ndarray,
        y: np.ndarray,
    ) -> "CalibrationModel":
        """
        Fit Platt scaling logistic regression.

        Parameters:
            X: feature matrix
            y: binary outcomes
        """

        self.model.fit(X, y)
        return self

    # ============================================================
    # Prediction
    # ============================================================

    def predict_confidence(
        self,
        score_vector: np.ndarray,
        signal_coverage: float,
    ) -> float:
        """
        Returns calibrated probability that the compatibility
        prediction is reliable.
        """

        features = self._feature_vector(
            score_vector,
            signal_coverage,
        ).reshape(1, -1)

        probability = self.model.predict_proba(features)[0, 1]

        return float(np.clip(probability, 0.0, 1.0))

    # ============================================================
    # Reliability Diagram Data
    # ============================================================

    def calibration_plot_data(
        self,
        predicted_probs: np.ndarray,
        true_labels: np.ndarray,
        bins: int = 10,
    ) -> Dict[str, List[float]]:
        """
        Returns calibration curve data.

        Output:
            {
                "bin_centers": [...],
                "predicted": [...],
                "observed": [...],
                "counts": [...]
            }
        """

        predicted_probs = np.asarray(predicted_probs)
        true_labels = np.asarray(true_labels)

        bin_edges = np.linspace(0.0, 1.0, bins + 1)

        predicted_means = []
        observed_freqs = []
        bin_centers = []
        counts = []

        for i in range(bins):
            lower = bin_edges[i]
            upper = bin_edges[i + 1]

            if i == bins - 1:
                mask = (
                    (predicted_probs >= lower)
                    & (predicted_probs <= upper)
                )
            else:
                mask = (
                    (predicted_probs >= lower)
                    & (predicted_probs < upper)
                )

            if np.sum(mask) == 0:
                continue

            probs_bin = predicted_probs[mask]
            labels_bin = true_labels[mask]

            predicted_means.append(float(np.mean(probs_bin)))
            observed_freqs.append(float(np.mean(labels_bin)))
            bin_centers.append(float((lower + upper) / 2.0))
            counts.append(int(np.sum(mask)))

        return {
            "bin_centers": bin_centers,
            "predicted": predicted_means,
            "observed": observed_freqs,
            "counts": counts,
        }

    # ============================================================
    # Serialization
    # ============================================================

    def to_json(self) -> str:
        """
        Serialize model to JSON.
        """

        payload = {
            "n_dimensions": self.n_dimensions,
            "coef": self.model.coef_.tolist(),
            "intercept": self.model.intercept_.tolist(),
            "classes": self.model.classes_.tolist(),
        }

        return json.dumps(payload)

    @classmethod
    def from_json(cls, data: str) -> "CalibrationModel":
        """
        Deserialize model from JSON.
        """

        payload = json.loads(data)

        model = LogisticRegression()

        model.classes_ = np.asarray(payload["classes"])
        model.coef_ = np.asarray(payload["coef"])
        model.intercept_ = np.asarray(payload["intercept"])

        calibration_model = cls(
            model=model,
            n_dimensions=payload["n_dimensions"],
        )

        return calibration_model


# ============================================================
# Example Usage
# ============================================================

if __name__ == "__main__":
    calibration_model = CalibrationModel.from_synthetic_data()

    team_scores = np.array([
        0.82,
        0.77,
        0.71,
        0.69,
        0.88,
        0.73,
        0.79,
        0.91,
    ])

    coverage = 0.75

    calibrated_confidence = calibration_model.predict_confidence(
        score_vector=team_scores,
        signal_coverage=coverage,
    )

    print("Calibrated Confidence:", calibrated_confidence)

    serialized = calibration_model.to_json()

    restored_model = CalibrationModel.from_json(serialized)

    restored_confidence = restored_model.predict_confidence(
        score_vector=team_scores,
        signal_coverage=coverage,
    )

    print("Restored Confidence:", restored_confidence)

# GitSyntropy: Engineering Team Compatibility Scoring System Evaluation Report

## 1. Executive Summary

This report provides a comprehensive evaluation of the GitSyntropy system, a machine learning model designed to score the compatibility of engineering teams. The system analyzes GitHub behavioral data and self-reported psychometric profiles to generate a compatibility score, aiming to predict team cohesion and performance.

The primary purpose of GitSyntropy is to offer a data-driven approach to team formation and to identify potential areas of friction within existing teams. The system scores teams across eight behavioral dimensions, each weighted from 1 to 8, culminating in a total compatibility score ranging from 0 to 36. This score is categorized as "poor" (<12), "fair" (12-20), "good" (20-28), or "excellent" (>28).

This evaluation covers the system's data sources, known biases and limitations, calibration, sensitivity to data sparsity, fairness considerations, and a comparison against baseline methods. The report concludes with recommendations for the responsible and effective production use of GitSyntropy. The intended audience for this document includes machine learning engineers, engineering managers, and Chief Technology Officers involved in the review and deployment of the system.

## 2. Dataset Description

The GitSyntropy model is trained and evaluated on data from two primary sources: publicly available GitHub repositories and self-reported psychometric assessments.

### 2.1. GitHub Behavioral Data

This data stream provides insights into the work habits and collaboration patterns of engineers. The following specific signals are extracted:

*   **Commit Timestamps:** Used to infer developer chronotypes (e.g., "early bird," "night owl") through K-Means clustering. This helps in assessing temporal alignment within a team.
*   **Pull Request (PR) Activity:** This includes metrics such as PR creation and review frequency, time to merge, and discussion length. These signals are proxies for coding velocity, review quality, and communication patterns.

### 2.2. Psychometric Profiling

This dataset consists of self-reported assessments from engineers on a 1-5 Likert scale. These assessments cover personality traits and work style preferences relevant to team dynamics.

### 2.3. Data Preprocessing and Imputation

Missing data is a common challenge. For any behavioral dimension where data is unavailable for a team member, the system imputes the value at the dimension's midpoint (50th percentile). This approach is a conservative measure to avoid unduly penalizing or rewarding teams for a lack of data.

### 2.4. Confidence Score

To provide transparency about the robustness of the compatibility score, a "confidence score" is calculated for each evaluation. This is determined by the ratio of observed signals to the total possible signals for a given team. A lower confidence score indicates that the compatibility score is based on a more limited dataset and should be interpreted with greater caution.

### 2.5. Known Biases in Public GitHub Data

Public GitHub data is not a perfect representation of all software development activity. Several known biases can influence the model's predictions:

*   **Open-Source vs. Corporate Skew:** Public data is dominated by open-source projects, which may have different collaboration norms and developer motivations compared to private, corporate projects.
*   **"Celebrity" Developer Effect:** Highly active and visible developers in popular open-source projects can skew the data, creating an unrepresentative sample of typical engineering behavior.

## 3. Known Biases and Limitations

It is crucial to acknowledge the inherent biases and limitations of the GitSyntropy system to ensure its responsible application.

### 3.1. UTC Timestamp Skew

Commit timestamps are recorded in UTC. While this provides a standardized measure, it can obscure the actual local working hours of developers in different time zones. The K-Means clustering for chronotype analysis may incorrectly group individuals due to this lack of localized time information.

### 3.2. Bot and Non-Individual Accounts

The system may inadvertently analyze the activity of bot accounts or organizational accounts, misinterpreting automated actions as individual developer behavior. This can lead to inaccurate representations of team dynamics.

### 3.3. Corporate vs. OSS Profiles

The collaboration patterns observed in open-source projects may not directly translate to a corporate environment. OSS developers often have more autonomy and different incentive structures, which can affect their GitHub behavior.

### 3.4. Survivorship Bias

The analysis is based on the activity of developers who remain active on GitHub. This introduces a survivorship bias, as the data does not include information on developers who have left the platform or specific projects. The reasons for their departure could be highly relevant to team compatibility.

## 4. Calibration Analysis

A well-calibrated model produces predictions that align with the true likelihood of an outcome. In the context of GitSyntropy, this means that a team with a "good" compatibility score should, on average, perform better than a team with a "poor" score.

## 5. Sensitivity Analysis

This analysis examines how the system's compatibility scores change in response to variations in the input data, particularly under conditions of data sparsity.

### 5.1. Impact of Data Sparsity

The confidence score is a key indicator of data availability. The following table illustrates a hypothetical scenario of how compatibility scores might fluctuate as the percentage of available data changes.

| Percentage of Available Data | Average Change in Score |
| ---------------------------- | ----------------------- |
| 100%                         | 0                       |
| 75%                          | +/- 2 points            |
| 50%                          | +/- 5 points            |
| 25%                          | +/- 10 points           |

This hypothetical data suggests that as data becomes more sparse, the potential for score volatility increases significantly. This underscores the importance of interpreting scores with low confidence with extreme caution.

## 6. Fairness Considerations

Ensuring fairness is paramount to the ethical deployment of any system that evaluates individuals. For GitSyntropy, we consider two key aspects of fairness:

### 6.1. Demographic Parity

This principle requires that the distribution of compatibility scores is similar across different demographic groups (e.g., gender, ethnicity). It is crucial to audit the system to ensure that no protected group is systematically receiving lower scores.

### 6.2. Equal Opportunity

Equal opportunity in this context means that for any given level of "true" team compatibility, the system should assign similar scores regardless of the demographic composition of the team. This prevents the system from penalizing certain groups for behaviors that would be scored differently in another group.

To address these concerns, regular fairness audits of the model's predictions against demographic data (where ethically and legally permissible to collect) are essential.

## 7. Comparison with Baselines

To understand the value of GitSyntropy, its performance should be compared against simpler, existing methods for team formation and evaluation.

### 7.1. Baseline Methods

*   **Random Assignment:** Teams are formed without any specific compatibility considerations. This serves as a "null hypothesis" to test if GitSyntropy provides any lift over chance.
*   **Peer Feedback Surveys:** Qualitative and quantitative feedback from team members is a common method for assessing team health and compatibility.

### 7.2. Comparative Performance

This table highlights the potential advantages of GitSyntropy in terms of scalability and objectivity. However, its predictive accuracy relative to these baselines needs to be rigorously validated through A/B testing and longitudinal studies.

## 8. Recommendations for Production Use

Based on this evaluation, the following recommendations are provided for the production deployment of the GitSyntropy system:

1.  **Use as a Guide, Not a Mandate:** GitSyntropy scores should be used as one input among many in the team formation process. They should not be the sole determinant of team composition.
2.  **Emphasize the Confidence Score:** Users must be trained to consider the confidence score alongside the compatibility score. Decisions should not be based on low-confidence predictions.
3.  **Regular Auditing for Bias and Fairness:** Continuous monitoring and auditing of the system's predictions for bias and fairness are critical. This includes analyzing score distributions across different demographic groups.
4.  **A/B Testing and Validation:** Before full-scale deployment, conduct A/B tests to compare the performance of teams formed with GitSyntropy's guidance against teams formed using traditional methods.
5.  **User Feedback and Iteration:** Establish a feedback loop with engineering managers and teams to gather qualitative data on the system's effectiveness and to identify areas for improvement.
6.  **Transparency in a Production Environment:** When used, be transparent with teams about how the system works, what data it uses, and its limitations. This can help build trust and mitigate concerns about being "judged by an algorithm."
